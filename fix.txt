### Phase 1: Role Suitability Indicator

Display a visual indicator (percentage or badge) on each player card on the pitch to show how well a player fits their currently assigned tactical role, based on `role_suitability` sent from the backend.

#### 1. Front-End Rendering (`tactics_pitch.js` / `free_formation.js`)

```javascript
// Renders the player card on the pitch with a dynamic suitability badge
function renderPlayerCardOnPitch(player) {
    const suitability = player.role_suitability || 0; // 0 to 100%
    
    // Determine badge color based on suitability score
    let badgeClass = 'bg-danger';
    if (suitability >= 80) badgeClass = 'bg-success';
    else if (suitability >= 60) badgeClass = 'bg-warning text-dark';

    return `
        <div class="player-pitch-card" data-player-id="${player.id}">
            <div class="player-name">${player.short_name}</div>
            <div class="player-role-selector">
                <select onchange="updatePlayerRole('${player.id}', this.value)" class="form-select form-select-sm">
                    ${renderRoleOptions(player.position, player.selected_role)}
                </select>
            </div>
            <!-- Role Suitability Badge -->
            <div class="role-suitability-bar mt-1">
                <span class="badge ${badgeClass}" style="font-size: 0.7rem;">
                    Suitability: ${suitability}%
                </span>
            </div>
        </div>
    `;
}

```

#### 2. JavaScript Logic (`tactics_center.js`)

```javascript
let currentEditingPlayerId = null;
const AVAILABLE_PIS = [
    { id: 'shoot_more', label: 'Shoot More Often' },
    { id: 'dribble_more', label: 'Dribble More' },
    { id: 'stay_wide', label: 'Stay Wide' },
    { id: 'mark_tighter', label: 'Mark Tighter' },
    { id: 'get_further_forward', label: 'Get Further Forward' }
];

function openPIModal(playerId, currentPIs = []) {
    currentEditingPlayerId = playerId;
    const container = document.getElementById('piOptionsContainer');
    container.innerHTML = '';

    AVAILABLE_PIS.forEach(pi => {
        const isChecked = currentPIs.includes(pi.id) ? 'checked' : '';
        container.innerHTML += `
            <div class="form-check form-check-inline me-3">
                <input class="form-check-input pi-checkbox" type="checkbox" value="${pi.id}" id="pi_${pi.id}" ${isChecked}>
                <label class="form-check-label" for="pi_${pi.id}">${pi.label}</label>
            </div>
        `;
    });

    // Enforce max limit of 3 instructions
    document.querySelectorAll('.pi-checkbox').forEach(cb => {
        cb.addEventListener('change', () => {
            const checkedCount = document.querySelectorAll('.pi-checkbox:checked').length;
            if (checkedCount > 3) {
                cb.checked = false;
                alert('You can select a maximum of 3 individual instructions per player.');
            }
        });
    });

    new bootstrap.Modal(document.getElementById('playerInstructionsModal')).show();
}

function savePlayerInstructions() {
    const selectedPIs = Array.from(document.querySelectorAll('.pi-checkbox:checked')).map(cb => cb.value);
    
    // Store selected instructions in JS memory state
    if (tacticsData.slots[currentEditingPlayerId]) {
        tacticsData.slots[currentEditingPlayerId].pi = selectedPIs;
    }
    
    bootstrap.Modal.getInstance(document.getElementById('playerInstructionsModal')).hide();
}

```

### Phase 3: Display Player Traits & Personality Badges

Surface key attributes (e.g., Leader, Long Shots, Tries Through Balls) directly on the tactical card on the pitch to help managers make informed decisions.

#### 1. Rendering Traits (`tactics_pitch.js`)

```javascript
function renderPlayerTraitsBadges(traits = [], personality = '') {
    let html = '<div class="traits-container d-flex justify-content-center gap-1 mt-1">';
    
    if (personality === 'Leader') {
        html += `<span class="badge bg-warning text-dark" title="Leader"><i class="bi bi-star-fill"></i> C</span>`;
    }
    
    traits.forEach(trait => {
        if (trait === 'long_shots') {
            html += `<span class="badge bg-info" title="Long Shots">🎯</span>`;
        } else if (trait === 'tries_through_balls') {
            html += `<span class="badge bg-primary" title="Tries Through Balls">📐</span>`;
        }
    });
    
    html += '</div>';
    return html;
}

```

### Phase 5: Full Tactical Payload Payload Integration

Connect all UI elements into a single `fetch` request that collects tactical style, individual instructions, set-piece takers, and line-ups, sending them to `/tactics/save`.

#### 1. API Request Logic (`tactics_center.js`)

```javascript
async function saveCompleteTactics() {
    // 1. Gather team tactical instructions
    const payload = {
        formation: document.getElementById('formationSelect').value,
        mentality: document.getElementById('mentalitySelect').value,
        passing_style: document.getElementById('passingStyleSelect').value,
        tempo: document.getElementById('tempoSelect').value,
        width: document.getElementById('widthSelect').value,
        defensive_line: document.getElementById('defensiveLineSelect').value,
        pressing_line: document.getElementById('pressingLineSelect').value,
        counter_press: document.getElementById('counterPressCheck').checked,
        counter_attack: document.getElementById('counterAttackCheck').checked,
        
        // 2. Gather set-piece routines
        set_pieces: {
            penalty: document.getElementById('penaltyTakerSelect').value,
            freekick_direct: document.getElementById('directFKSelect').value,
            corner_left: document.getElementById('cornerLeftSelect').value,
            corner_right: document.getElementById('cornerRightSelect').value,
            long_throw: document.getElementById('longThrowSelect').value
        },
        
        // 3. Gather pitch lineup, assigned roles, and individual instructions (PIs)
        slots: tacticsData.slots // Array of objects: { player_id, role, position, pi: [...] }
    };

    // 4. Send payload to Flask endpoint
    try {
        const response = await fetch('/tactics/save', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(payload)
        });

        const result = await response.json();
        if (result.success) {
            alert('Tactics and player instructions saved successfully!');
        } else {
            alert('Failed to save tactics: ' + result.message);
        }
    } catch (error) {
        console.error('Error saving tactics:', error);
    }
}

```
